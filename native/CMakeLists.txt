cmake_minimum_required(VERSION 3.20)
project(ginseng_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GINSENG_SANITIZE "Build standalone sanitizer exercise" OFF)
if(GINSENG_SANITIZE)
  add_executable(kernel_sanitize src/sanitize.cpp)
  target_compile_options(kernel_sanitize PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-fast-math -ffp-contract=off)
  target_link_options(kernel_sanitize PRIVATE -fsanitize=address,undefined)
else()
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core src/core.cpp)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/core.cpp" CORE_HASH)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/kernels.hpp" KERNEL_HASH)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" CMAKE_HASH)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/pyproject.toml" PACKAGE_HASH)
  target_compile_definitions(_core PRIVATE GINSENG_CMAKE_HASH="${CMAKE_HASH}" GINSENG_PACKAGE_HASH="${PACKAGE_HASH}" GINSENG_BUILD_TYPE="${CMAKE_BUILD_TYPE}" GINSENG_COMPILER_FLAGS="${CMAKE_CXX_FLAGS}")
  target_compile_definitions(_core PRIVATE GINSENG_SOURCE_HASH="${CORE_HASH}" GINSENG_KERNEL_HASH="${KERNEL_HASH}")
  if(MSVC)
    target_compile_options(_core PRIVATE /fp:strict)
  else()
    target_compile_options(_core PRIVATE -fno-fast-math -ffp-contract=off)
  endif()
  install(TARGETS _core LIBRARY DESTINATION ginseng_native)
endif()
